type TradeProposal struct {
    TenantID       string          `json:"tenant_id"`
    IdempotencyKey string          `json:"idempotency_key"`
    Ticker         string          `json:"ticker"`
    Action         TradeAction     `json:"action"` // "BUY" | "SELL" | "HOLD"
    Confidence     float64         `json:"confidence"`
    Agents         []AgentDecision `json:"agent_deliberations"`
}

// Fail Closed: 拒绝任何未知字段与尾随非法数据
func DecodeTradeStrict(r io.Reader) (*TradeProposal, error) {
    dec := json.NewDecoder(r)
    dec.DisallowUnknownFields()
    var p TradeProposal
    if err := dec.Decode(&p); err != nil { return nil, ErrMaliciousPayload }
    return &p, nil
}
